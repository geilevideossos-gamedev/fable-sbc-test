# ---- VoxHub CM4 carrier (added by os/build-image.sh) -------------------------
# Only *stock* Raspberry Pi overlays are used here on purpose: HAOS wipes
# /overlays on every OS update but keeps config.txt, so this block survives OTA.
[cm4]
# USB 2.0 host on the CM4 (disabled by default) -> USB2514B hub -> USB-A ports
otg_mode=1
[all]
# I2S audio: ADAU7002 (2x PDM mics) capture + MAX98357A playback, amp SD_MODE on GPIO16
dtoverlay=googlevoicehat-soundcard
# Zigbee/Thread radio (MGM240P) on UART4: TX=GPIO8 RX=GPIO9 CTS=GPIO10 RTS=GPIO11 -> /dev/ttyAMA1
dtoverlay=uart4,ctsrts
# (radio reset GPIO24 is active high with a pull-down on the board, bootloader strap GPIO25 is pulled up - nothing to configure)
# Status LEDs (active high, through N-FETs): /sys/class/leds/voxhub-green, voxhub-blue
dtoverlay=gpio-led,gpio=5,label=voxhub-green,trigger=none
dtoverlay=gpio-led,gpio=6,label=voxhub-blue,trigger=none
# Hardware mute switch sense (high = muted) -> input event KEY_MICMUTE (248)
dtoverlay=gpio-key,gpio=27,label=voxhub-mute,keycode=248,active_low=0,gpio_pull=off
# I2C1 (GPIO2/3): USB-PD controller status (CYPD3177 HPI @ 0x08)
dtparam=i2c_arm=on
# Serial console on the debug header (GPIO14/15) - uncomment for bring-up
#enable_uart=1
# ---- end VoxHub ---------------------------------------------------------------
